import tkinter as tk from tkinter import messagebox import random class AppliDefisGigantesque: def __init__(self, root): self.root = root self.root.title("⚠️ CHALLENGE CONTROL GIGA EDITION ⚠️") self.root.geometry("500x350") self.root.configure(bg="#1e1e2e") self.root.attributes("-topmost", True) self.score = 0 self.defis_realises = 0 self.max_defis_par_session = 10 # Demande s'il veut continuer tous les 10 défis self.temps_restant = 5 # Génération automatique de milliers de défis self.LISTE_DEFIS = self.generer_2000_defis() # --- INTERFACE --- self.label_titre = tk.Label(root, text="CHALLENGE CONTROL", font=("Arial", 16, "bold"), fg="#ff0055", bg="#1e1e2e") self.label_titre.pack(pady=10) self.label_chrono = tk.Label(root, text="Initialisation...", font=("Arial", 12), fg="#cdd6f4", bg="#1e1e2e") self.label_chrono.pack(pady=5) self.label_defi = tk.Label(root, text="Préparation de la giga liste...", font=("Arial", 11, "italic"), fg="#f9e2af", bg="#313244", wraplength=450, height=5, width=50) self.label_defi.pack(pady=15) self.frame_boutons = tk.Frame(root, bg="#1e1e2e") self.frame_boutons.pack(pady=5) self.btn_reussi = tk.Button(self.frame_boutons, text="✅ Réussi (+10 pts)", font=("Arial", 10, "bold"), fg="#ffffff", bg="#a6e3a1", command=self.defi_reussi) self.btn_echoue = tk.Button(self.frame_boutons, text="❌ Échoué", font=("Arial", 10, "bold"), fg="#ffffff", bg="#f38ba8", command=self.defi_echoue) self.label_score = tk.Label(root, text="Score : 0 pts | Défis : 0", font=("Arial", 11, "bold"), fg="#89b4fa", bg="#1e1e2e") self.label_score.pack(side="bottom", pady=10) self.actualiser_chrono() def generer_2000_defis(self): # Des verbes, des actions et des contraintes pour créer des milliers de combinaisons actions = [ "ne dis plus le mot 'OUI' ni 'NON'", "parle uniquement en chuchotant", "place le mot 'Banane' de façon naturelle", "parle avec un accent super bizarre", "interdiction absolue de sauter", "fais un bruit de klaxon", "répète la fin de toutes tes phrases", "parle comme un robot", "ferme un œil", "garde tes mains au-dessus de la tête", "chante au lieu de parler", "ajoute 'wesh' ou 'du coup' partout", "fais semblant d'être hyper fatigué", "n'utilise pas ta touche préférée", "parle uniquement en faisant des rimes", "cite un streamer connu" ] contraintes = [ "pendant les 2 prochaines minutes !", "pendant les 60 prochaines secondes.", "chaque fois que tu touches ton clavier !", "à chaque fois que tu gagnes un point.", "pendant tout le prochain round !", "jusqu'à la fin du chrono.", "dès que tu ouvres une nouvelle page !", "à la prochaine phrase que tu dis." ] defis_uniques = set() # On fait tourner la machine pour combiner tout ça jusqu'à avoir plus de 2000 défis while len(defis_uniques) < 2500: act = random.choice(actions) con = random.choice(contraintes) # Petit ajout aléatoire pour varier encore plus bonus = random.choice(["", " (Et fais-le sérieusement !)", " (Double score si réussi !)", " (C'est un giga défi !)"]) defis_uniques.add(f"👉 {act} {con}{bonus}") return list(defis_uniques) def actualiser_chrono(self): if self.temps_restant > 0: self.label_chrono.config(text=f"Prochain défi dans : {self.temps_restant}s", fg="#cdd6f4") self.temps_restant -= 1 self.root.after(1000, self.actualiser_chrono) else: self.declencher_defi() def declencher_defi(self): self.label_chrono.config(text="⚠️ NOUVEAU DÉFI ! ⚠️", fg="#ff0055") # Pioche un défi dans notre giga liste defi_choisi = random.choice(self.LISTE_DEFIS) self.label_defi.config(text=defi_choisi, font=("Arial", 11, "bold"), fg="#ff0055") self.btn_reussi.pack(side="left", padx=10) self.btn_echoue.pack(side="right", padx=10) def defi_reussi(self): self.score += 10 self.defis_realises += 1 self.verifier_continuer() def defi_echoue(self): self.defis_realises += 1 self.verifier_continuer() def verifier_continuer(self): self.label_score.config(text=f"Score : {self.score} pts | Défis : {self.defis_realises}") self.btn_reussi.pack_forget() self.btn_echoue.pack_forget() # Si on a atteint la limite de la session, on demande s'il veut continuer if self.defis_realises % self.max_defis_par_session == 0: reponse = messagebox.askyesno("Continuer ?", f"Tu as fait {self.defis_realises} défis ! Veux-tu d'autres défis ?") if not reponse: self.label_defi.config(text="Session terminée ! Quel giga score !", fg="#a6e3a1") self.label_chrono.config(text="STOP", fg="#f38ba8") return # Arrête l'application ici self.label_defi.config(text="Bien joué ! En attente du prochain...", font=("Arial", 11, "italic"), fg="#f9e2af") self.temps_restant = 5 # Temps avant le prochain self.actualiser_chrono() if __name__ == "__main__": fenetre = tk.Tk() app = AppliDefisGigantesque(fenetre) fenetre.mainloop()
